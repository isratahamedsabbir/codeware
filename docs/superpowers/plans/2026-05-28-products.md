# Products Feature Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a full product catalog system — three DB tables, two Eloquent models, two Livewire admin components, four REST API controllers, and a seeder for the five product categories.

**Architecture:** Mirror the existing CMS pattern exactly. Livewire admin (same as Posts/Categories) for managing products in the Laravel admin panel. Two sets of REST API endpoints under `/api/v1/` — public (no auth, locale-aware) and admin (Sanctum + `can:access-admin` gate). Product images link to the existing `media_library` table via a `product_media` pivot.

**Tech Stack:** Laravel 11, Livewire 3, Flux UI, Spatie Translatable, Pest PHP, SQLite (tests), PHP 8.5 at `C:\Users\User\.config\herd\bin\php85\php.exe` (use `php artisan` if Herd adds it to PATH, otherwise use the full path).

```php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\BelongsToMany;
use Illuminate\Database\Eloquent\SoftDeletes;
use Illuminate\Support\Str;
use Spatie\Translatable\HasTranslations;

class Product extends Model
{
    use HasFactory, SoftDeletes, HasTranslations;

    public array $translatable = ['name', 'excerpt', 'description', 'usage_instructions', 'seo_title', 'seo_description'];

    protected $fillable = [
        'product_category_id', 'name', 'slug', 'excerpt', 'description',
        'specifications', 'benefits', 'usage_instructions',
        'featured_image', 'datasheet_url', 'status', 'is_featured',
        'sort_order', 'seo_title', 'seo_description',
    ];

    protected $casts = [
        'specifications' => 'array',
        'benefits'       => 'array',
        'is_featured'    => 'boolean',
        'sort_order'     => 'integer',
    ];

    protected static function booted(): void
    {
        static::saving(function (Product $product) {
            if (empty($product->slug)) {
                $name = is_array($product->name)
                    ? ($product->name['en'] ?? reset($product->name))
                    : $product->name;
                $product->slug = Str::slug($name);
            }
        });
    }

    public function category(): BelongsTo
    {
        return $this->belongsTo(ProductCategory::class, 'product_category_id');
    }

    public function gallery(): BelongsToMany
    {
        return $this->belongsToMany(MediaLibrary::class, 'product_media')
            ->withPivot('sort_order')
            ->orderBy('product_media.sort_order');
    }

    public function scopePublished(Builder $query): Builder
    {
        return $query->where('status', 'published');
    }

    public function scopeFeatured(Builder $query): Builder
    {
        return $query->where('is_featured', true);
    }
}
```

- [ ] **Step 1: Create the Livewire component**

Create `app/Livewire/Admin/ProductCategories/Index.php`:

```php
<?php

namespace App\Livewire\Admin\ProductCategories;

use App\Models\ProductCategory;
use Livewire\Attributes\Validate;
use Livewire\Component;
use Livewire\WithPagination;

class Index extends Component
{
    use WithPagination;

    public string $search = '';

    #[Validate('required|string|max:255')]
    public string $name_en = '';

    #[Validate('nullable|string|max:255')]
    public string $name_bn = '';

    #[Validate('nullable|string|max:255')]
    public string $slug = '';

    #[Validate('nullable|string')]
    public string $description_en = '';

    #[Validate('nullable|string')]
    public string $description_bn = '';

    #[Validate('nullable|string|max:50')]
    public string $icon = '';

    #[Validate('nullable|integer|min:0')]
    public int $sort_order = 0;

    public ?int $editingId = null;
    public ?int $deletingId = null;

    public function updatedSearch(): void
    {
        $this->resetPage();
    }

    public function openCreate(): void
    {
        $this->reset('name_en', 'name_bn', 'slug', 'description_en', 'description_bn', 'icon', 'sort_order', 'editingId');
        $this->sort_order = 0;
        $this->resetValidation();
        $this->dispatch('open-modal', name: 'product-category-form');
    }

    public function openEdit(int $id): void
    {
        $cat = ProductCategory::findOrFail($id);
        $this->editingId    = $id;
        $this->name_en      = $cat->getTranslation('name', 'en', false) ?? '';
        $this->name_bn      = $cat->getTranslation('name', 'bn', false) ?? '';
        $this->slug         = $cat->slug;
        $this->description_en = $cat->getTranslation('description', 'en', false) ?? '';
        $this->description_bn = $cat->getTranslation('description', 'bn', false) ?? '';
        $this->icon         = $cat->icon ?? '';
        $this->sort_order   = $cat->sort_order;
        $this->resetValidation();
        $this->dispatch('open-modal', name: 'product-category-form');
    }

    public function save(): void
    {
        $this->validate();

        $data = [
            'name'        => array_filter(['en' => $this->name_en, 'bn' => $this->name_bn]),
            'slug'        => $this->slug,
            'description' => array_filter(['en' => $this->description_en, 'bn' => $this->description_bn]) ?: null,
            'icon'        => $this->icon ?: null,
            'sort_order'  => $this->sort_order,
        ];

        if ($this->editingId) {
            ProductCategory::findOrFail($this->editingId)->update($data);
        } else {
            ProductCategory::create($data);
        }

        $this->reset('name_en', 'name_bn', 'slug', 'description_en', 'description_bn', 'icon', 'sort_order', 'editingId');
        $this->dispatch('close-modal', name: 'product-category-form');
    }

    public function confirmDelete(int $id): void
    {
        $this->deletingId = $id;
        $this->dispatch('open-modal', name: 'product-category-delete');
    }

    public function delete(): void
    {
        if ($this->deletingId) {
            ProductCategory::findOrFail($this->deletingId)->delete();
            $this->deletingId = null;
        }
        $this->dispatch('close-modal', name: 'product-category-delete');
    }

    public function render()
    {
        return view('livewire.admin.product-categories.index', [
            'categories' => ProductCategory::query()
                ->when($this->search, fn ($q) => $q
                    ->where('name->en', 'like', "%{$this->search}%")
                    ->orWhere('name->bn', 'like', "%{$this->search}%"))
                ->withCount('products')
                ->orderBy('sort_order')
                ->paginate(15),
        ])->layout('layouts.admin', ['title' => 'Product Categories']);
    }
}
```

```php
<?php

namespace App\Livewire\Admin\Products;

use App\Models\MediaLibrary;
use App\Models\Product;
use App\Models\ProductCategory;
use Livewire\Attributes\Computed;
use Livewire\Attributes\On;
use Livewire\Attributes\Validate;
use Livewire\Component;
use Livewire\WithPagination;

class Index extends Component
{
    use WithPagination;

    public string $search = '';
    public string $statusFilter = '';

    // Basic
    #[Validate('required|string|max:255')]
    public string $name_en = '';
    #[Validate('nullable|string|max:255')]
    public string $name_bn = '';
    #[Validate('nullable|string|max:255')]
    public string $slug = '';
    #[Validate('nullable|integer|exists:product_categories,id')]
    public ?int $product_category_id = null;
    #[Validate('in:draft,published')]
    public string $status = 'draft';
    public bool $is_featured = false;
    public int $sort_order = 0;

    // Content
    #[Validate('nullable|string')]
    public string $excerpt_en = '';
    #[Validate('nullable|string')]
    public string $excerpt_bn = '';
    #[Validate('nullable|string')]
    public string $description_en = '';
    #[Validate('nullable|string')]
    public string $description_bn = '';
    #[Validate('nullable|string')]
    public string $usage_instructions_en = '';
    #[Validate('nullable|string')]
    public string $usage_instructions_bn = '';
    public array $specifications = [];
    public array $benefits = [];

    // Media
    public string $featured_image = '';
    public string $datasheet_url = '';
    public array $galleryItems = [];
    public string $mediaTarget = '';

    // SEO
    #[Validate('nullable|string|max:255')]
    public string $seo_title_en = '';
    #[Validate('nullable|string|max:255')]
    public string $seo_title_bn = '';
    #[Validate('nullable|string')]
    public string $seo_description_en = '';
    #[Validate('nullable|string')]
    public string $seo_description_bn = '';

    public ?int $editingId = null;
    public ?int $deletingId = null;

    public function updatedSearch(): void { $this->resetPage(); }
    public function updatedStatusFilter(): void { $this->resetPage(); }

    #[Computed]
    public function productCategories()
    {
        return ProductCategory::orderBy('sort_order')->get();
    }

    public function openCreate(): void
    {
        $this->resetForm();
        $this->resetValidation();
        $this->dispatch('open-modal', name: 'product-form');
    }

    public function openEdit(int $id): void
    {
        $product = Product::with('gallery')->findOrFail($id);
        $this->editingId = $id;

        $this->name_en      = $product->getTranslation('name', 'en', false) ?? '';
        $this->name_bn      = $product->getTranslation('name', 'bn', false) ?? '';
        $this->slug         = $product->slug;
        $this->product_category_id = $product->product_category_id;
        $this->status       = $product->status;
        $this->is_featured  = (bool) $product->is_featured;
        $this->sort_order   = $product->sort_order;

        $this->excerpt_en   = $product->getTranslation('excerpt', 'en', false) ?? '';
        $this->excerpt_bn   = $product->getTranslation('excerpt', 'bn', false) ?? '';
        $this->description_en = $product->getTranslation('description', 'en', false) ?? '';
        $this->description_bn = $product->getTranslation('description', 'bn', false) ?? '';
        $this->usage_instructions_en = $product->getTranslation('usage_instructions', 'en', false) ?? '';
        $this->usage_instructions_bn = $product->getTranslation('usage_instructions', 'bn', false) ?? '';

        $this->specifications = array_map(fn ($spec) => [
            'label_en' => $spec['label']['en'] ?? '',
            'label_bn' => $spec['label']['bn'] ?? '',
            'value_en' => $spec['value']['en'] ?? '',
            'value_bn' => $spec['value']['bn'] ?? '',
        ], $product->specifications ?? []);

        $enBenefits = ($product->benefits ?? [])['en'] ?? [];
        $bnBenefits = ($product->benefits ?? [])['bn'] ?? [];
        $this->benefits = array_map(fn ($en, $bn) => [
            'text_en' => $en ?? '',
            'text_bn' => $bn ?? '',
        ], $enBenefits, $bnBenefits) ?: [];

        $this->featured_image = $product->featured_image ?? '';
        $this->datasheet_url  = $product->datasheet_url ?? '';
        $this->galleryItems   = $product->gallery->map(fn ($m) => [
            'id'  => $m->id,
            'url' => $m->url,
            'alt' => $m->alt_text ?? '',
        ])->values()->toArray();

        $this->seo_title_en        = $product->getTranslation('seo_title', 'en', false) ?? '';
        $this->seo_title_bn        = $product->getTranslation('seo_title', 'bn', false) ?? '';
        $this->seo_description_en  = $product->getTranslation('seo_description', 'en', false) ?? '';
        $this->seo_description_bn  = $product->getTranslation('seo_description', 'bn', false) ?? '';

        $this->resetValidation();
        $this->dispatch('open-modal', name: 'product-form');
    }

    public function openFeaturedImagePicker(): void
    {
        $this->mediaTarget = 'featured_image';
        $this->dispatch('open-modal', name: 'media-picker');
    }

    public function openGalleryPicker(): void
    {
        $this->mediaTarget = 'gallery';
        $this->dispatch('open-modal', name: 'media-picker');
    }

    public function openDatasheetPicker(): void
    {
        $this->mediaTarget = 'datasheet';
        $this->dispatch('open-modal', name: 'media-picker');
    }

    public function removeGalleryItem(int $index): void
    {
        array_splice($this->galleryItems, $index, 1);
    }

    #[On('media-selected')]
    public function onMediaSelected(int $id, string $url, string $alt = ''): void
    {
        match ($this->mediaTarget) {
            'featured_image' => $this->featured_image = $url,
            'datasheet'      => $this->datasheet_url = $url,
            'gallery'        => $this->galleryItems[] = ['id' => $id, 'url' => $url, 'alt' => $alt],
            default          => null,
        };
        $this->mediaTarget = '';
    }

    public function addSpecification(): void
    {
        $this->specifications[] = ['label_en' => '', 'label_bn' => '', 'value_en' => '', 'value_bn' => ''];
    }

    public function removeSpecification(int $index): void
    {
        array_splice($this->specifications, $index, 1);
    }

    public function addBenefit(): void
    {
        $this->benefits[] = ['text_en' => '', 'text_bn' => ''];
    }

    public function removeBenefit(int $index): void
    {
        array_splice($this->benefits, $index, 1);
    }

    public function save(): void
    {
        $this->validate();

        $data = [
            'product_category_id' => $this->product_category_id,
            'name'   => array_filter(['en' => $this->name_en, 'bn' => $this->name_bn]),
            'slug'   => $this->slug,
            'status' => $this->status,
            'is_featured' => $this->is_featured,
            'sort_order'  => $this->sort_order,
            'excerpt'     => array_filter(['en' => $this->excerpt_en, 'bn' => $this->excerpt_bn]) ?: null,
            'description' => array_filter(['en' => $this->description_en, 'bn' => $this->description_bn]) ?: null,
            'usage_instructions' => array_filter(['en' => $this->usage_instructions_en, 'bn' => $this->usage_instructions_bn]) ?: null,
            'featured_image' => $this->featured_image ?: null,
            'datasheet_url'  => $this->datasheet_url ?: null,
            'seo_title'      => array_filter(['en' => $this->seo_title_en, 'bn' => $this->seo_title_bn]) ?: null,
            'seo_description'=> array_filter(['en' => $this->seo_description_en, 'bn' => $this->seo_description_bn]) ?: null,
            'specifications' => $this->specifications
                ? array_map(fn ($s) => [
                    'label' => array_filter(['en' => $s['label_en'], 'bn' => $s['label_bn']]),
                    'value' => array_filter(['en' => $s['value_en'], 'bn' => $s['value_bn']]),
                ], $this->specifications)
                : null,
            'benefits' => $this->benefits
                ? ['en' => array_column($this->benefits, 'text_en'), 'bn' => array_column($this->benefits, 'text_bn')]
                : null,
        ];

        if ($this->editingId) {
            $product = Product::findOrFail($this->editingId);
            $product->update($data);
        } else {
            $product = Product::create($data);
        }

        $gallerySync = collect($this->galleryItems)
            ->mapWithKeys(fn ($item, $idx) => [$item['id'] => ['sort_order' => $idx]])
            ->all();
        $product->gallery()->sync($gallerySync);

        $this->resetForm();
        $this->dispatch('close-modal', name: 'product-form');
    }

    public function confirmDelete(int $id): void
    {
        $this->deletingId = $id;
        $this->dispatch('open-modal', name: 'product-delete');
    }

    public function delete(): void
    {
        if ($this->deletingId) {
            Product::findOrFail($this->deletingId)->delete();
            $this->deletingId = null;
        }
        $this->dispatch('close-modal', name: 'product-delete');
    }

    private function resetForm(): void
    {
        $this->reset(
            'name_en', 'name_bn', 'slug', 'product_category_id', 'status',
            'is_featured', 'sort_order', 'excerpt_en', 'excerpt_bn',
            'description_en', 'description_bn', 'usage_instructions_en', 'usage_instructions_bn',
            'specifications', 'benefits', 'featured_image', 'datasheet_url',
            'galleryItems', 'mediaTarget', 'seo_title_en', 'seo_title_bn',
            'seo_description_en', 'seo_description_bn', 'editingId'
        );
        $this->status = 'draft';
    }

    public function render()
    {
        return view('livewire.admin.products.index', [
            'products' => Product::query()
                ->with('category')
                ->when($this->search, fn ($q) => $q
                    ->where('name->en', 'like', "%{$this->search}%")
                    ->orWhere('name->bn', 'like', "%{$this->search}%")
                    ->orWhere('slug', 'like', "%{$this->search}%"))
                ->when($this->statusFilter, fn ($q) => $q->where('status', $this->statusFilter))
                ->orderBy('sort_order')
                ->latest()
                ->paginate(20),
        ])->layout('layouts.admin', ['title' => 'Products']);
    }
}
```

```php
<?php

namespace App\Http\Controllers\Api\V1;

use App\Http\Controllers\Controller;
use App\Models\Product;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class ProductController extends Controller
{
    private function resolveLocale(Request $request): string
    {
        $locale = $request->query('locale');
        return in_array($locale, ['en', 'bn'], true) ? $locale : 'en';
    }

    public function index(Request $request): JsonResponse
    {
        $locale  = $this->resolveLocale($request);
        $perPage = min((int) $request->query('per_page', 12), 100);

        $products = Product::published()
            ->with('category')
            ->orderBy('sort_order')
            ->when($request->query('category'), fn ($q, $slug) =>
                $q->whereHas('category', fn ($c) => $c->where('slug', $slug)))
            ->when($request->query('search'), fn ($q, $search) =>
                $q->where("name->{$locale}", 'like', "%{$search}%"))
            ->when($request->query('featured') === '1', fn ($q) =>
                $q->where('is_featured', true))
            ->paginate($perPage);

        return response()->json([
            'data' => $products->map(fn ($p) => $this->formatProduct($p, $locale)),
            'meta' => [
                'current_page' => $products->currentPage(),
                'last_page'    => $products->lastPage(),
                'per_page'     => $products->perPage(),
                'total'        => $products->total(),
            ],
        ]);
    }

    public function show(Request $request, string $slug): JsonResponse
    {
        $locale = $this->resolveLocale($request);

        $product = Product::published()
            ->with(['category', 'gallery'])
            ->where('slug', $slug)
            ->firstOrFail();

        $related = Product::published()
            ->where('product_category_id', $product->product_category_id)
            ->where('id', '!=', $product->id)
            ->orderBy('sort_order')
            ->limit(4)
            ->get();

        return response()->json([
            'data' => $this->formatProduct($product, $locale, withDetail: true, related: $related),
        ]);
    }

    private function formatProduct(Product $product, string $locale, bool $withDetail = false, $related = null): array
    {
        $data = [
            'id'            => $product->id,
            'slug'          => $product->slug,
            'name'          => $product->getTranslation('name', $locale, useFallbackLocale: true),
            'excerpt'       => $product->getTranslation('excerpt', $locale, useFallbackLocale: true),
            'featured_image'=> $product->featured_image,
            'datasheet_url' => $product->datasheet_url,
            'is_featured'   => $product->is_featured,
            'sort_order'    => $product->sort_order,
            'seo_title'     => $product->getTranslation('seo_title', $locale, useFallbackLocale: true),
            'seo_description'=> $product->getTranslation('seo_description', $locale, useFallbackLocale: true),
            'category'      => $product->category ? [
                'id'   => $product->category->id,
                'slug' => $product->category->slug,
                'name' => $product->category->getTranslation('name', $locale, useFallbackLocale: true),
            ] : null,
        ];

        if ($withDetail) {
            $data['description'] = $product->getTranslation('description', $locale, useFallbackLocale: true);
            $data['usage_instructions'] = $product->getTranslation('usage_instructions', $locale, useFallbackLocale: true);
            $data['specifications'] = collect($product->specifications ?? [])->map(fn ($spec) => [
                'label' => $spec['label'][$locale] ?? $spec['label']['en'] ?? '',
                'value' => $spec['value'][$locale] ?? $spec['value']['en'] ?? '',
            ])->values();
            $benefitsArr = $product->benefits ?? [];
            $data['benefits'] = $benefitsArr[$locale] ?? $benefitsArr['en'] ?? [];
            $data['gallery'] = $product->gallery->map(fn ($m) => [
                'id'         => $m->id,
                'url'        => $m->url,
                'alt'        => $m->alt_text ?? '',
                'sort_order' => $m->pivot->sort_order,
            ])->values();
            $data['related_products'] = $related->map(fn ($p) => $this->formatProduct($p, $locale))->values();
        }

        return $data;
    }
}
```
